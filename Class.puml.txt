' Relationships with Multiplicity

Customer "1" -- "*" Account : has
Account "1" -- "*" Transaction : contains
Customer "1" -- "*" Loan : applies
Loan "*" -- "1" Manager : reviewed
Employee "*" -- "1" Manager : manages
Employee "1" -- "*" Transaction : processes
Account "1" -- "*" Employee : serviced
Transaction "*" -- "1" Customer : belongs
